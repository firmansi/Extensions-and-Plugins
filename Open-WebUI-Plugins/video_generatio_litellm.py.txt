    def _stitch_clips(self, clips tuple[bytes, ...], music bytes  None, narration bytes  None) - bytes

                with open(path, wb) as output
                    output.write(clip)
                clip_paths.append(path)
            manifest = os.path.join(temporary_directory, clips.txt)
            with open(manifest, w, encoding=utf-8) as output
                output.writelines(ffile '{path}'n for path in clip_paths)
            stitched = os.path.join(temporary_directory, stitched.mp4)
            concatenate = subprocess.run(
                [self.valves.FFMPEG_BIN, -y, -f, concat, -safe, 0, -i, manifest, -c, copy, stitched],
                capture_output=True,
                check=False,

            if concatenate.returncode != 0
                details = concatenate.stderr.decode(utf-8, errors=replace)[-1000]
                raise RuntimeError(fCould not stitch video clips with ffmpeg {details})

                music_path = os.path.join(temporary_directory, music.mp3)
                narration_path = os.path.join(temporary_directory, narration.mp3)

    async def _generate_clip(
        self,
        session aiohttp.ClientSession,
        payload dict[str, object],
        selected_model str,
        emitter EventEmitter  None,
    ) - bytes
        submission_payload = payload.copy()
        async with session.post(
            f{self._base_url()}v1videos, headers=self._headers(), json=submission_payload
        ) as response
            if response.status in {200, 201, 202}
                created Final[object] = await response.json()
            else
                error_body Final = await response.text()
                if not self._unsupported_resolution_error(response.status, error_body, submission_payload)
                    raise RuntimeError(fVideo submission failed with HTTP {response.status} {error_body})
                submission_payload.pop(resolution)
                await self._emit(emitter, Selected resolution is unsupported; retrying with the model default...)
                async with session.post(
                    f{self._base_url()}v1videos, headers=self._headers(), json=submission_payload
                ) as retry
                    if retry.status not in {200, 201, 202}
                        raise RuntimeError(fVideo submission failed with HTTP {retry.status} {await retry.text()})
                    created = await retry.json()
        if not isinstance(created, dict) or not isinstance(created.get(id), str)
            raise RuntimeError(LiteLLM did not return a video job id)
        video_id Final = created[id]
        started Final = time.monotonic()
        while True
            elapsed Final = int(time.monotonic() - started)
            if elapsed  self.valves.MAX_TIMEOUT_SECONDS
                raise RuntimeError(fVideo generation timed out after {self.valves.MAX_TIMEOUT_SECONDS} seconds)
            await self._emit(emitter, fGenerating video with {selected_model} ({elapsed}s elapsed)...)
            await asyncio.sleep(self.valves.POLL_INTERVAL_SECONDS)
            async with session.get(f{self._base_url()}v1videos{video_id}, headers=self._headers()) as response
                if response.status != 200
                    raise RuntimeError(
                        fVideo status check failed with HTTP {response.status} {await response.text()}
                    )
                status_body Final[object] = await response.json()
            if not isinstance(status_body, dict)
                raise RuntimeError(LiteLLM returned an invalid video status response)
            if status_body.get(status) == completed
                break
            if status_body.get(status) in {failed, cancelled, expired}
                raise RuntimeError(fVideo generation failed {status_body.get('error', 'unknown error')})
        async with session.get(f{self._base_url()}v1videos{video_id}content, headers=self._headers()) as response
            if response.status != 200
                raise RuntimeError(fVideo download failed with HTTP {response.status} {await response.text()})
            return await response.read()
